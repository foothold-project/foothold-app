cmake_minimum_required(VERSION 3.20)
project(foothold_start_runner LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(foothold_core src/core.cpp src/runner.cpp)
target_include_directories(foothold_core PUBLIC include)
target_compile_options(foothold_core PRIVATE -Wall -Wextra -Wpedantic)
include(CTest)
if(BUILD_TESTING)
    add_executable(core_test tests/core_test.cpp)
    target_link_libraries(core_test PRIVATE foothold_core)
    add_test(NAME core_contract COMMAND core_test)
    add_executable(runner_test tests/runner_test.cpp)
    target_link_libraries(runner_test PRIVATE foothold_core)
    add_test(NAME runner_transitions COMMAND runner_test)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(Threads REQUIRED)
        add_executable(mode_worker_test tests/mode_worker_test.cpp src/mode_worker.cpp src/thread_schedule.cpp)
        target_link_libraries(mode_worker_test PRIVATE foothold_core Threads::Threads)
        target_compile_options(mode_worker_test PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME mode_rpc_worker COMMAND mode_worker_test)
        set_tests_properties(mode_rpc_worker PROPERTIES TIMEOUT 8)
        add_executable(depth_receiver_test tests/depth_receiver_test.cpp src/depth_receiver.cpp)
        target_link_libraries(depth_receiver_test PRIVATE foothold_core)
        target_compile_options(depth_receiver_test PRIVATE -Wall -Wextra -Wpedantic)
        add_test(NAME depth_datagrams COMMAND depth_receiver_test)
    endif()
    add_executable(command_replay tests/command_replay.cpp)
    target_link_libraries(command_replay PRIVATE foothold_core)
endif()

option(FOOTHOLD_WITH_ORT "Build the ONNX Runtime integration" OFF)
set(FOOTHOLD_VECTOR_BUNDLE "" CACHE PATH "Directory containing the canonical test vectors")
if(FOOTHOLD_WITH_ORT)
    find_path(ONNXRUNTIME_INCLUDE_DIR onnxruntime_cxx_api.h REQUIRED)
    find_library(ONNXRUNTIME_LIBRARY onnxruntime REQUIRED)
    find_package(OpenSSL REQUIRED COMPONENTS Crypto)
    add_library(foothold_ort src/ort_policy.cpp)
    target_include_directories(foothold_ort SYSTEM PRIVATE "${ONNXRUNTIME_INCLUDE_DIR}")
    target_link_libraries(foothold_ort PUBLIC foothold_core PRIVATE "${ONNXRUNTIME_LIBRARY}" OpenSSL::Crypto)
    target_compile_options(foothold_ort PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_TESTING AND FOOTHOLD_VECTOR_BUNDLE)
        find_package(Python3 REQUIRED COMPONENTS Interpreter)
        add_executable(vector_replay tests/vector_replay.cpp)
        target_link_libraries(vector_replay PRIVATE foothold_ort)
        add_test(NAME onnx_vectors COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/vector_replay.py"
            "$<TARGET_FILE:vector_replay>" "${FOOTHOLD_VECTOR_BUNDLE}")
    endif()
endif()

option(FOOTHOLD_WITH_UNITREE "Build the Unitree SDK2 integration" OFF)
if(FOOTHOLD_WITH_UNITREE)
    find_package(unitree_sdk2 REQUIRED CONFIG)
    add_library(foothold_unitree src/unitree_robot.cpp)
    target_link_libraries(foothold_unitree PUBLIC foothold_core PRIVATE unitree_sdk2)
    target_compile_options(foothold_unitree PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(FOOTHOLD_WITH_ORT)
    find_package(Threads REQUIRED)
    add_executable(foothold_start src/main.cpp src/trace.cpp src/policy_worker.cpp src/depth_receiver.cpp
        src/thread_schedule.cpp src/depth_sampler.cpp src/mode_worker.cpp)
    target_link_libraries(foothold_start PRIVATE foothold_ort Threads::Threads)
    target_compile_options(foothold_start PRIVATE -Wall -Wextra -Wpedantic)
    if(BUILD_TESTING AND FOOTHOLD_VECTOR_BUNDLE AND NOT CMAKE_CROSSCOMPILING)
        add_test(NAME runner_cli_dry_run COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/runner_cli.py"
            "$<TARGET_FILE:foothold_start>" "${FOOTHOLD_VECTOR_BUNDLE}")
        set_tests_properties(runner_cli_dry_run PROPERTIES TIMEOUT 25)
        add_test(NAME runner_schedule COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/runner_schedule.py"
            "$<TARGET_FILE:foothold_start>" "${FOOTHOLD_VECTOR_BUNDLE}")
        set_tests_properties(runner_schedule PROPERTIES TIMEOUT 25 SKIP_RETURN_CODE 77)
    endif()
    if(FOOTHOLD_WITH_UNITREE)
        target_compile_definitions(foothold_start PRIVATE FOOTHOLD_UNITREE)
        target_link_libraries(foothold_start PRIVATE foothold_unitree)
    endif()
endif()
